### 3.2 `sky.gdshader`

```glsl
shader_type sky;

global uniform vec3 sun_dir;
global uniform float star_visibility;

uniform vec3 nebula_a : source_color = vec3(0.10, 0.06, 0.22);
uniform vec3 nebula_b : source_color = vec3(0.02, 0.14, 0.24);
uniform vec3 space_base : source_color = vec3(0.004, 0.006, 0.016);
uniform vec3 sun_color : source_color = vec3(1.0, 0.92, 0.78);

float hash13(vec3 p) {
	p = fract(p * 0.1031);
	p += dot(p, p.zyx + 31.32);
	return fract((p.x + p.y) * p.z);
}

vec3 hash33(vec3 p) {
	p = fract(p * vec3(0.1031, 0.1030, 0.0973));
	p += dot(p, p.yxz + 33.33);
	return fract((p.xxy + p.yxx) * p.zyx);
}

float value_noise(vec3 p) {
	vec3 i = floor(p);
	vec3 f = fract(p);
	f = f * f * (3.0 - 2.0 * f);
	float n000 = hash13(i);
	float n100 = hash13(i + vec3(1.0, 0.0, 0.0));
	float n010 = hash13(i + vec3(0.0, 1.0, 0.0));
	float n110 = hash13(i + vec3(1.0, 1.0, 0.0));
	float n001 = hash13(i + vec3(0.0, 0.0, 1.0));
	float n101 = hash13(i + vec3(1.0, 0.0, 1.0));
	float n011 = hash13(i + vec3(0.0, 1.0, 1.0));
	float n111 = hash13(i + vec3(1.0, 1.0, 1.0));
	float x00 = mix(n000, n100, f.x);
	float x10 = mix(n010, n110, f.x);
	float x01 = mix(n001, n101, f.x);
	float x11 = mix(n011, n111, f.x);
	return mix(mix(x00, x10, f.y), mix(x01, x11, f.y), f.z);
}

float fbm(vec3 p) {
	float sum = 0.0;
	float amp = 0.5;
	for (int i = 0; i < 4; i++) {
		sum += value_noise(p) * amp;
		p *= 2.03;
		amp *= 0.5;
	}
	return sum;
}

// 一层星点：把方向放大到网格里，每格最多一颗星，位置和亮度随机
float star_layer(vec3 dir, float scale, float density) {
	vec3 p = dir * scale;
	vec3 cell = floor(p);
	vec3 local = fract(p) - 0.5;
	vec3 offset = (hash33(cell) - 0.5) * 0.7;
	float present = step(1.0 - density, hash13(cell + 17.0));
	float d = length(local - offset);
	float size = 0.06 + hash13(cell + 5.0) * 0.08;
	return present * smoothstep(size, 0.0, d) * (0.4 + 0.6 * hash13(cell + 9.0));
}

void sky() {
	if (AT_CUBEMAP_PASS) {
		COLOR = space_base;
	} else {
		vec3 dir = EYEDIR;
		float n = fbm(dir * 2.2);
		float band = fbm(dir * 5.0 + vec3(3.1, 0.0, 1.7));
		vec3 nebula = mix(nebula_a, nebula_b, band) * smoothstep(0.35, 0.8, n);
		float stars = star_layer(dir, 180.0, 0.10) + star_layer(dir, 420.0, 0.06) * 0.6;
		vec3 col = space_base + nebula + vec3(stars) * 2.5;
		float sun = smoothstep(0.9994, 0.9997, dot(dir, sun_dir));
		float halo = pow(max(dot(dir, sun_dir), 0.0), 600.0) * 0.6;
		col = col * star_visibility + sun_color * (sun * 8.0 + halo);
		COLOR = col;
	}
}
```

太阳圆盘不受 `star_visibility` 影响，白天和夜晚都画；其余部分在白天由大气壳盖住。

## 第四步：卡通光照

### 4.1 `toon_light.gdshaderinc`

所有受光的材质都 `#include` 这个文件，保证全场的明暗规则一致。

```glsl
// 卡通光照：亮面、半亮、暗面三档，交界处保留一点柔和
// 暗面只剩环境色（由 SkyDirector 设为蓝紫），所以暗部偏冷而不是发黑
void light() {
	float ndl = dot(NORMAL, LIGHT);
	float lit = smoothstep(0.0, 0.05, ndl);
	float tier = mix(0.72, 1.0, smoothstep(0.30, 0.36, ndl));
	float shadow = smoothstep(0.25, 0.75, ATTENUATION);
	DIFFUSE_LIGHT += LIGHT_COLOR / PI * lit * tier * shadow;
}
```

`LIGHT_COLOR` 已经乘过 π，这里除回去，与 Godot 文档示例的写法一致。定义了 `light()` 以后不再计算高光，这正是卡通风格要的效果。

## 第五步：地形高度函数

### 5.1 `terrain_noise.gd`

```gdscript
class_name TerrainNoise
extends RefCounted

# 所有高度单位为米，相对海平面；输入为单位方向向量
const MAX_HEIGHT: float = 2600.0

var radius: float
var _continent: FastNoiseLite
var _mountain: FastNoiseLite
var _hills: FastNoiseLite
var _detail: FastNoiseLite


func _init(seed: int, planet_radius: float) -> void:
	radius = planet_radius
	_continent = _make(seed, FastNoiseLite.TYPE_SIMPLEX_SMOOTH, 1.0 / 14000.0, FastNoiseLite.FRACTAL_FBM, 4)
	_mountain = _make(seed + 1, FastNoiseLite.TYPE_SIMPLEX_SMOOTH, 1.0 / 3800.0, FastNoiseLite.FRACTAL_RIDGED, 5)
	_hills = _make(seed + 2, FastNoiseLite.TYPE_SIMPLEX_SMOOTH, 1.0 / 700.0, FastNoiseLite.FRACTAL_FBM, 3)
	_detail = _make(seed + 3, FastNoiseLite.TYPE_SIMPLEX_SMOOTH, 1.0 / 90.0, FastNoiseLite.FRACTAL_FBM, 2)


func _make(seed: int, type: FastNoiseLite.NoiseType, frequency: float, fractal: FastNoiseLite.FractalType, octaves: int) -> FastNoiseLite:
	var noise := FastNoiseLite.new()
	noise.seed = seed
	noise.noise_type = type
	noise.frequency = frequency
	noise.fractal_type = fractal
	noise.fractal_octaves = octaves
	return noise


func height(dir: Vector3) -> float:
	var p: Vector3 = dir * radius
	var continent: float = _continent.get_noise_3dv(p)
	# 大陆值决定海陆：低于 -0.05 为海，高于 0.25 为陆，中间是海岸
	var land: float = smoothstep(-0.05, 0.25, continent)
	# 山脉只长在大陆内部
	var mountain_mask: float = smoothstep(0.30, 0.65, continent)
	var ridge: float = (_mountain.get_noise_3dv(p) + 1.0) * 0.5
	var mountain: float = ridge * ridge * mountain_mask * 2400.0
	var hills: float = _hills.get_noise_3dv(p) * 90.0 * land
	var detail: float = _detail.get_noise_3dv(p) * 4.0
	return lerpf(-420.0, 40.0, land) + mountain + hills + detail
```

### 5.2 线程安全

`height()` 会在多个工作线程中同时调用。`FastNoiseLite` 的取值函数只读不写，可以并发调用。如果出现崩溃，或者结果偶尔不对，就给每个工作任务 `duplicate()` 一份 `TerrainNoise` 中的噪声对象，并在开发记录里写明。

### 6.1 常量与立方体六个面

```gdscript
extends Node3D

const FACES: Array = [
```

### 6.2 立方体到球面

用"球化立方体"映射，比直接 `normalized()` 的网格分布更均匀：

```gdscript
static func cube_to_sphere(p: Vector3) -> Vector3:
	var x2: float = p.x * p.x
	var y2: float = p.y * p.y
	var z2: float = p.z * p.z
	return Vector3(
		p.x * sqrt(1.0 - y2 * 0.5 - z2 * 0.5 + y2 * z2 / 3.0),
		p.y * sqrt(1.0 - z2 * 0.5 - x2 * 0.5 + z2 * x2 / 3.0),
		p.z * sqrt(1.0 - x2 * 0.5 - y2 * 0.5 + x2 * y2 / 3.0))
```

输入是立方体表面上的点（某一维为 ±1），输出已经是单位向量。

### 6.3 四叉树节点

```gdscript
class TerrainNode:
	extends RefCounted
	var face: int
	var u0: float                    # 该块在面上的起点，面坐标范围 [-1, 1]
	var v0: float
	var size: float                  # 面坐标跨度，根节点为 2
	var depth: int
	var edge_m: float                # 近似边长（米）
	var center_local: Vector3        # 块中心在星球局部坐标中的位置
	var children: Array[TerrainNode] = []
	var mesh_instance: MeshInstance3D = null
	var water_instance: MeshInstance3D = null
	var collider: StaticBody3D = null
	var task_id: int = -1
	var result: Dictionary = {}
	var wants_collision: bool = false
```

- `edge_m = radius * size * 0.785`。根节点的跨度为 2，约等于四分之一圆周。
- `center_local = cube_to_sphere(n + a * uc + b * vc) * (radius + noise.height(dir))`，其中 `uc`、`vc` 是块中心的面坐标。

### 6.4 每 0.15 秒更新一次细节层级

```gdscript
func _update_lod() -> void:
	var cam: Vector3 = to_local(_camera.global_position)
	for root: TerrainNode in _roots:
		_visit(root, cam)


func _visit(node: TerrainNode, cam: Vector3) -> void:
	var distance: float = cam.distance_to(node.center_local)
	var should_split: bool = node.depth < MAX_DEPTH and distance < node.edge_m * SPLIT_FACTOR and _above_horizon(node, cam)
	var should_merge: bool = distance > node.edge_m * MERGE_FACTOR
	if should_split:
		if node.children.is_empty():
			_create_children(node)
		var all_ready: bool = true
		for child: TerrainNode in node.children:
			if child.mesh_instance == null:
				_request_build(child)
				all_ready = false
		# 四个子块全部就绪后才隐藏父块，过渡期间不会露洞
		if all_ready:
			_set_visible(node, false)
			for child: TerrainNode in node.children:
				_visit(child, cam)
	else:
		if node.mesh_instance == null:
			_request_build(node)
		elif should_merge or node.children.is_empty():
			_set_visible(node, true)
			_free_children(node)
	node.wants_collision = node.depth == MAX_DEPTH and distance < COLLISION_RANGE
```

`_above_horizon`：块中心低于相机地平线的不再细分，背面的山不会浪费细节。

```gdscript
func _above_horizon(node: TerrainNode, cam: Vector3) -> bool:
	var cam_r: float = cam.length()
	var horizon: float = sqrt(maxf(cam_r * cam_r - radius * radius, 0.0))
	var reach: float = sqrt(pow(radius + TerrainNoise.MAX_HEIGHT, 2.0) - radius * radius)
	return cam.distance_to(node.center_local) < horizon + reach + node.edge_m
```

### 6.5 后台生成（`terrain_chunk_builder.gd`）

只做纯计算，**不创建任何节点或资源**，返回一个 Dictionary：

```gdscript
class_name TerrainChunkBuilder
extends RefCounted


static func build(noise: TerrainNoise, n: Vector3, a: Vector3, b: Vector3, u0: float, v0: float, size: float, grid: int, skirt: float) -> Dictionary:
	var radius: float = noise.radius
	var count: int = grid + 3
	var pos := PackedVector3Array()
	pos.resize(count * count)
	var heights := PackedFloat32Array()
	heights.resize(count * count)
	# 多采一圈边界，用来算边缘法线，保证相邻块接缝处的光照连续
	for j: int in count:
		for i: int in count:
			var u: float = u0 + size * float(i - 1) / float(grid)
			var v: float = v0 + size * float(j - 1) / float(grid)
			var dir: Vector3 = PlanetTerrain.cube_to_sphere(n + a * u + b * v)
			var h: float = noise.height(dir)
			heights[j * count + i] = h
			pos[j * count + i] = dir * (radius + h)
	var center: Vector3 = pos[(grid / 2 + 1) * count + grid / 2 + 1]
	# …继续：顶点、法线、颜色、索引、裙边、海面，见下文
```

接下来的处理：

- **顶点**：只取内部 `(grid + 1) × (grid + 1)` 个点，坐标**减去 `center`**，即相对块中心。块节点的 `position` 设为 `center`。这样顶点数值很小，精度不会出问题。
- **法线**：`(pos[i+1, j] - pos[i-1, j]).cross(pos[i, j+1] - pos[i, j-1]).normalized()`。因为 a × b 指向外，这样算出的法线朝外。
- **颜色**：`COLOR.r = clamp(h / MAX_HEIGHT * 0.5 + 0.5, 0, 1)`，`COLOR.g = 1 - normal.dot(dir)`，即坡度。
- **索引**：Godot 以**顺时针**为正面。对每个格子 (i, j)，四个角是 v00、v10、v01、v11（i 为 a 方向，j 为 b 方向）。三角形为 `v00, v01, v10` 和 `v10, v01, v11`。
- **裙边**：四条边上的每个点，再复制一份并沿 `-dir` 方向下压 `skirt` 米，与原边连成一圈竖直的条带，用来遮住相邻块精度不同造成的裂缝。`skirt = edge_m * 0.03`。裙边的三角形**正反两个方向各写一次**，从哪一侧都看得到。
- **海面**：如果最低高度 < 0，用同样的网格、高度 0 再生成一组海面顶点（法线 = dir），放在返回值的 `water` 键里。
- **碰撞面**：仅在 `with_faces` 为 true 时生成 `PackedVector3Array`，每个三角形 3 个点（相对块中心）。

### 6.6 提交与回收任务

```gdscript
func _request_build(node: TerrainNode) -> void:
	if node.task_id != -1 or _running.size() >= MAX_TASKS:
		return
	var f: Array = FACES[node.face]
	node.task_id = WorkerThreadPool.add_task(func() -> void:
		node.result = TerrainChunkBuilder.build(_noise, f[0], f[1], f[2], node.u0, node.v0, node.size, GRID, node.edge_m * 0.03))
	_running.append(node)
```

在 `_process` 中：

1. 遍历 `_running`。对 `WorkerThreadPool.is_task_completed(id)` 为 true 的节点，调用 `wait_for_task_completion(id)` 回收任务，**每个任务都必须回收**，然后放进上传队列。
2. 每帧最多上传 `MAX_UPLOADS_PER_FRAME` 块：用 `ArrayMesh.add_surface_from_arrays` 创建网格，新建 `MeshInstance3D`，`position = result.center`，材质为 `terrain.gdshader`。海面同理，使用 `water.gdshader`。
3. 节点在任务进行期间已经被合并掉的话，回收任务后直接丢弃结果。
4. `wants_collision` 为 true 且还没有碰撞体的节点：用 `ConcavePolygonShape3D.set_faces()` 建碰撞。碰撞面数据要求重新生成一次（带 `with_faces`），也在后台线程中完成。离开范围后释放碰撞体。

## 第七步：大气壳与海

### 7.1 `atmosphere.gdshader`

大气壳是一个半径为 `radius + atmosphere_height` 的球体（`SphereMesh`，128 段 × 64 环），作为 `Planet` 的子节点。它同时负责三件事：从太空看到的光晕、地面上的天空颜色、远山的淡化。

```glsl
shader_type spatial;
render_mode unshaded, cull_front, depth_draw_never, depth_test_disabled, blend_mix, fog_disabled, shadows_disabled;

global uniform vec3 sun_dir;

uniform sampler2D depth_tex : hint_depth_texture, filter_nearest;
uniform vec3 planet_center;
uniform float planet_radius;
uniform float atmo_radius;
uniform float density = 6.0;
uniform float falloff = 4.0;
uniform vec3 day_color : source_color = vec3(0.52, 0.74, 1.0);
uniform vec3 sunset_color : source_color = vec3(1.0, 0.56, 0.38);
uniform vec3 night_color : source_color = vec3(0.03, 0.04, 0.10);

vec2 ray_sphere(vec3 ro, vec3 rd, vec3 c, float r) {
	vec3 oc = ro - c;
	float b = dot(oc, rd);
	float k = dot(oc, oc) - r * r;
	float h = b * b - k;
	if (h < 0.0) {
		return vec2(1e20, -1e20);
	}
	h = sqrt(h);
	return vec2(-b - h, -b + h);
}

void fragment() {
	float depth = textureLod(depth_tex, SCREEN_UV, 0.0).r;
	vec4 view = INV_PROJECTION_MATRIX * vec4(SCREEN_UV * 2.0 - 1.0, depth, 1.0);
	view.xyz /= view.w;
	float scene_dist = length(view.xyz);
	vec3 ro = CAMERA_POSITION_WORLD;
	vec3 rd = normalize((INV_VIEW_MATRIX * vec4(view.xyz, 0.0)).xyz);
	vec2 shell = ray_sphere(ro, rd, planet_center, atmo_radius);
	float t0 = max(shell.x, 0.0);
	float t1 = min(shell.y, scene_dist);
	vec2 ground = ray_sphere(ro, rd, planet_center, planet_radius);
	if (ground.x > 0.0) {
		t1 = min(t1, ground.x);
	}
	if (t1 <= t0) {
		discard;
	}
	// 沿视线分 8 段积分：越低越稠密，越靠前的大气挡住后面的
	float thickness = atmo_radius - planet_radius;
	float seg = (t1 - t0) / 8.0;
	float optical = 0.0;
	vec3 sum = vec3(0.0);
	float weight = 0.0;
	for (int i = 0; i < 8; i++) {
		vec3 p = ro + rd * (t0 + seg * (float(i) + 0.5));
		vec3 up = normalize(p - planet_center);
		float h = clamp((length(p - planet_center) - planet_radius) / thickness, 0.0, 1.0);
		float d = exp(-h * falloff) * seg / thickness * density;
		float sun_h = dot(up, sun_dir);
		vec3 lit = mix(sunset_color, day_color, smoothstep(-0.02, 0.30, sun_h));
		lit = mix(night_color, lit, smoothstep(-0.22, 0.04, sun_h));
		float w = d * exp(-optical);
		sum += lit * w;
		weight += w;
		optical += d;
	}
	ALBEDO = sum / max(weight, 1e-5);
	ALPHA = clamp(1.0 - exp(-optical), 0.0, 1.0);
}
```

要点：

- 深度重建用 `vec4(SCREEN_UV * 2.0 - 1.0, depth, 1.0)`。这是 Godot 4.3 以后（反向 Z）Forward+ 渲染器的写法，**不要**再把深度做 `* 2 - 1` 的变换。
- `cull_front`：无论相机在球壳内还是球壳外，都画球的背面，所以两种情况下都能看到大气。
- `depth_test_disabled`：大气不被地形挡住，到地形的距离用深度图计算。
- `planet_center` 是**世界坐标**，原点跟随以后会变，所以由 `SkyDirector` 每帧更新。

### 7.2 `terrain.gdshader`

```glsl
shader_type spatial;
render_mode cull_back;

#include "res://shaders/toon_light.gdshaderinc"

global uniform float rim_strength;

uniform vec3 sand : source_color = vec3(0.86, 0.80, 0.64);
uniform vec3 grass : source_color = vec3(0.47, 0.66, 0.52);
uniform vec3 grass_far : source_color = vec3(0.55, 0.70, 0.62);
uniform vec3 rock : source_color = vec3(0.56, 0.52, 0.60);
uniform vec3 snow : source_color = vec3(0.95, 0.96, 1.0);
uniform float max_height = 2600.0;

varying float v_height;
varying float v_slope;

void vertex() {
	v_height = (COLOR.r * 2.0 - 1.0) * max_height;
	v_slope = COLOR.g;
}

void fragment() {
	vec3 c = mix(grass, grass_far, smoothstep(300.0, 1200.0, v_height));
	c = mix(sand, c, smoothstep(4.0, 22.0, v_height));
	c = mix(c, rock, smoothstep(0.22, 0.34, v_slope));
	float snow_line = smoothstep(1500.0, 1650.0, v_height) * (1.0 - smoothstep(0.30, 0.45, v_slope));
	c = mix(c, snow, snow_line);
	ALBEDO = c;
	ROUGHNESS = 1.0;
	float rim = pow(1.0 - clamp(dot(NORMAL, VIEW), 0.0, 1.0), 4.0);
	EMISSION = ALBEDO * rim * rim_strength;
}
```

## 第九步：世界原点跟随与玩家

### 9.1 `floating_origin.gd`

```gdscript
extends Node

# 玩家离原点超过阈值时整体平移宇宙，让附近物体的坐标始终保持小数值，避免单精度抖动
@export var threshold: float = 2000.0
@export var universe_path: NodePath
@export var player_path: NodePath

@onready var _universe: Node3D = get_node(universe_path)
@onready var _player: Node3D = get_node(player_path)


func _physics_process(_delta: float) -> void:
	var offset: Vector3 = _player.global_position
	if offset.length() > threshold:
		_universe.global_position -= offset
		_player.global_position -= offset


# 玩家在宇宙中的真实坐标，用于日志与验证
func true_position() -> Vector3:
	return _player.global_position - _universe.global_position
```

`FloatingOrigin` 节点在场景树中必须排在 `Player` **之前**，保证平移发生在玩家移动之前。

### 9.2 `player.tscn`


### 9.3 `player.gd` 的要点

**两种模式**：`WALK`（步行）和 `FLY`（飞行），按 F 切换。

**每个物理帧的开头**：

```gdscript
var planet: Planet = _nearest_planet()
var to_center: Vector3 = planet.global_position - global_position
var distance: float = to_center.length()
var up: Vector3 = -to_center / distance
var in_gravity: bool = distance < planet.radius * 3.0
if in_gravity:
	up_direction = up
	_align_to_up(up, delta)
```

**让身体的上方对齐重力方向**：

```gdscript
# 平滑转向，避免进入引力范围时镜头猛地一跳
func _align_to_up(up: Vector3, delta: float) -> void:
	var forward: Vector3 = -global_basis.z
	forward = (forward - up * forward.dot(up)).normalized()
	var z_axis: Vector3 = -forward
	var x_axis: Vector3 = up.cross(z_axis)
	var target := Basis(x_axis, up, z_axis).orthonormalized()
	global_basis = global_basis.slerp(target, clampf(delta * 8.0, 0.0, 1.0))
```

**鼠标视角**：水平转动是身体绕 `up` 旋转（`rotate(up, -dx * 0.0025)`）；俯仰是 `Head` 绕自身 X 轴旋转，限制在 ±89°。

**步行**：

- 输入方向投影到切平面：`(forward * y + right * x)`，其中 `forward` 和 `right` 都已去掉 `up` 分量。
- 速度：走 4 米/秒，跑（Shift）8 米/秒。
- 不在地面时，`velocity += -up * planet.gravity * delta`。
- 空格跳跃，起跳速度 5 米/秒，沿 `up` 方向。
- 调用 `move_and_slide()`。`up_direction` 已设为 `up`，所以地面判定在球面上同样有效。

**飞行**：

- 不受重力，W/S 沿镜头前方，A/D 横移，空格和 Ctrl 沿 `up` 升降。
- 速度随离地高度变化：`speed = clampf(altitude * 1.0, 12.0, 30000.0)`，Shift 乘以 4。离地面越近越慢，在太空中每秒可飞数十公里。
- `velocity = velocity.lerp(target, clampf(delta * 3.0, 0.0, 1.0))`，起步和停下都有惯性。
- **贴地保护**：`altitude = distance - planet.surface_radius(up)`。如果 `altitude < 1.2`，把玩家沿 `up` 推到地面以上 1.2 米。这样高速飞行时不会穿进地形，而且不依赖碰撞体。
- 离地 2 米以内按 F，切回步行。

**出生点**：星球表面一处陆地，离海岸 300 米以内，附近 8 公里内有 1200 米以上的山（找法同第十二步的"自动选点"）。面朝那座山。

## 第十一步：昼夜与 `SkyDirector`

`sky_director.gd` 每帧执行以下操作：

1. **太阳旋转**：太阳绕星球的自转轴（星球局部 Y 轴）转动，一天 20 分钟。按住 T 时加速 60 倍。`Sun` 的方向设为 `-sun_dir`，即光沿 `-Z` 射出，`global_basis.z == sun_dir`。
2. **相机所在位置**：最近的星球、`up`、海拔，以及 `sun_h = dot(up, sun_dir)`。
3. **两个系数**：
   - `day = smoothstep(-0.12, 0.18, sun_h)`；
   - `space = smoothstep(atmosphere_height * 0.5, atmosphere_height * 1.5, altitude)`。
4. **全局参数**：
   - `RenderingServer.global_shader_parameter_set("sun_dir", sun_dir)`；
   - `star_visibility = maxf(1.0 - day, space)`。
5. **太阳颜色**：`sunset = Color(1.0, 0.62, 0.42)`，`noon = Color(1.0, 0.97, 0.92)`，按 `smoothstep(0.0, 0.35, sun_h)` 插值；进入太空后逐渐变为 `noon`。亮度 `light_energy = lerp(0.0, 1.0, smoothstep(-0.08, 0.05, sun_h))`，在太空中恒为 1。
6. **环境色**（暗面的颜色）：
   - 白天 `Color(0.42, 0.45, 0.70)`，亮度 0.55；
   - 夜晚 `Color(0.10, 0.12, 0.28)`，亮度 0.35；
   - 太空 `Color(0.08, 0.08, 0.16)`，亮度 0.3。
   - 按 `day` 和 `space` 插值。
7. **大气材质**：每帧更新 `planet_center = planet.global_position`。

### 完成标准

按住 T，在地面上看完一个完整的昼夜：日落时天空和大地变暖，夜里星空出现、暗面变成深蓝，日出时反过来。整个过程没有突变。

## 第十三步：验证工具

`verify.gd`（自动加载）读取 `OS.get_cmdline_user_args()`。没有参数时什么都不做。

### 13.1 自动选点

截图和出生点共用同一个确定性的选点函数。种子固定，每次运行结果相同：

1. 用种子 `20261019` 生成 4000 个均匀分布的随机方向。
2. 依次检查：高度在 20～200 米（陆地低处），并且在 8 公里内沿某个方向（采样 16 个方向，每个方向每 500 米取一点）能找到高度 > 1200 米的点。
3. 第一个满足条件的点即为"远山点"，记下面朝远山的方向。

### 13.2 机位

| 名称 | 位置 | 朝向 | 太阳 |
| --- | --- | --- | --- |
| `s1_space` | 离星球中心 160 公里，太阳与相机方向夹角 60° | 看向星球中心，让气态巨星也在画面里 | 固定 |
| `s2_high` | 远山点上空 12 公里 | 看向远山方向的地平线，俯 8° | 太阳高度 35° |

| `s4_ground_sunset` | 同上 | 同上，太阳位于身后右侧 | 太阳高度 3° |
| `s5_ground_night` | 同上 | 抬头 35° | 太阳高度 −30° |
| `s6_shore` | 离远山点最近的海岸，眼高 1.62 米 | 看向海 | 太阳高度 25° |

"太阳高度"是指在机位所在点，太阳与地平面的夹角。由 `SkyDirector` 提供一个设置函数，直接把太阳放到要求的角度。

### 13.3 命令

- `--shot=<名称> --out=<目录>`：放好机位，等到地形队列为空后再过 120 帧，截图存为 `<名称>.png`。然后测 240 帧，把帧间隔、CPU、GPU 的平均值、P95、最大值写入 `<名称>.json`，之后退出。
- `--descent=<文件>`：从 `s1_space` 出发，沿直线飞到远山点地面，全程 60 秒，速度按飞行模式的规则计算。记录每一帧的帧间隔、CPU、GPU、当帧上传块数和海拔，写入 JSON。最后汇总：P99、超过 50 ms 的帧数，以及每个这样的帧的海拔和当时的块数。
- `--walk=<文件>`：在远山点向远山方向步行 40 秒，每 0.5 秒记录一次位置和海拔。判定：海拔从未低于 −0.3 米（没有掉进地下），40 秒内的水平位移 > 120 米（没有卡住）。

## 附录：常见问题

| 现象 | 原因 | 处理 |
| --- | --- | --- |
| 从太空看不到地形，或者只能看到背面 | 三角形绕序反了 | 按 6.5 节检查索引顺序；6.1 的 `assert` 必须通过 |
| 相邻地形块之间有细缝 | 没有生成裙边，或者裙边只写了一面 | 见 6.5 节的裙边 |
| 光照在地形块之间有接缝 | 法线没有用边界上多采的一圈来算 | 见 6.5 节，`count = grid + 3` |
| 地形块闪烁、反复出现又消失 | 细分和合并的阈值太接近 | 保持 `SPLIT_FACTOR` < `MERGE_FACTOR` |
| 降落时卡顿 | 在主线程生成地形，或者每帧上传太多块 | 生成必须走 `WorkerThreadPool`，上传每帧 ≤ 4 块 |
| 报警告说有任务未回收 | 没有对每个任务调用 `wait_for_task_completion` | 见 6.6 节 |
| 地面上看不到大气，或者大气盖住了一切 | 深度重建写法错误 | 见 7.1 节，不要对深度做 `* 2 - 1` |
| 走动时镜头抖动 | 原点跟随没有生效，或者执行顺序不对 | 见 9.1 节，节点顺序 |
| 着色器报错找不到 `sun_dir` | 全局参数没有在项目设置中声明 | 见 2.2 节 |
| 暗面一片漆黑 | 环境光来源没有设成 Color | 见 3.1 节 |
| 星星在白天也能看见 | `star_visibility` 没有更新 | 见第十一步 |
